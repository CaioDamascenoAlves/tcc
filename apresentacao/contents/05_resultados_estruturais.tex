% Slide: Análise de Confounders
\begin{frame}{Análise de Confounders}
    \begin{block}{Hipóteses Alternativas Testadas}
        \footnotesize
        Densidade F $>$ M poderia ser artefato de tamanho da rede, longevidade ou tipologia esportiva.
    \end{block}

    \vspace{0.2cm}

    \begin{block}{Regressão Múltipla: densidade $\sim$ Gênero + Tamanho + Longevidade + Tipologia}
        \begin{columns}[T]
            \begin{column}{0.54\textwidth}
                \centering
                \scriptsize
                \vspace{0.1cm}
                \begin{tabular}{lrrr}
                \toprule
                \textbf{Variável} & \textbf{Coef.} & \textbf{p-valor} & \textbf{Sig.} \\
                \midrule
                \textbf{Gênero (F=1)} & \textbf{-0.029} & \textbf{0.028} & \textbf{**} \\
                log(Atletas)          &  -0.009         &  0.582          & NS            \\
                log(Anos)             &  -0.113         &  0.078          & $\sim$        \\
                Tipo (Coletivo)       &  -0.012         &  0.974          & NS            \\
                \bottomrule
                \end{tabular}
                \vspace{0.05cm}
                \tiny $R^2 = 0.91$ \quad bootstrap $n=1000$ \quad \textbf{**} $p<0.05$, NS não significativo
            \end{column}
            \begin{column}{0.42\textwidth}
                \footnotesize
                \vspace{0.2cm}
                Apenas \textbf{gênero} é significativo (p = 0.028).\\[0.2cm]
                Tamanho (p = 0.582) e tipologia (p = 0.974) \textbf{não explicam} a diferença.\\[0.2cm]
                \textbf{R² = 0.91}: o modelo explica 91\% da variação de densidade.
            \end{column}
        \end{columns}
    \end{block}

    \vspace{0.2cm}

    \begin{alertblock}{Conclusão}
        \footnotesize\centering
        Densidade F $>$ M é \textbf{efeito genuíno e isolado}, não artefato metodológico
    \end{alertblock}
\end{frame}

% Slide 2 (C): Provocação — o limite da análise tradicional
\begin{frame}{O limite da análise tradicional}
    \begin{block}{O que uma tabela responde\ldots}
        \footnotesize\centering
        USA venceu \textbf{55\% dos ouros} no atletismo 100m em 120 anos \cite{Olympics100mWinners2024}
    \end{block}

    \vspace{0.3cm}

    \begin{alertblock}{O que ela não responde\ldots}
        \footnotesize
        \begin{itemize}
            \item Usain Bolt teria vencido Carl Lewis se competissem na mesma era?
            \item Quando exatamente essa dominância americana começou a ceder?
            \item Quem foi o atleta que conectou a era Lewis à era Bolt?
            \item Por que o Futebol F forma uma rede completamente diferente do Futebol M?
        \end{itemize}
    \end{alertblock}

    \vspace{0.3cm}

    \begin{center}
        \footnotesize
        Para responder isso, precisamos modelar \textbf{relações} --- não apenas resultados.
    \end{center}
\end{frame}

% Slide 2: Motivação e Contexto
\begin{frame}[shrink=10]{Motivação e Contexto}
    \begin{block}{120 anos de história olímpica (1896--2016)}
        \begin{itemize}
            \item \textbf{135.571 atletas} competindo em \textbf{66 modalidades}
            \item Análises tradicionais focam em \textit{contagens de medalhas por país}
            \item \textbf{O que vai além:} quem venceu quem? Por quanto tempo? Em que era?
        \end{itemize}
    \end{block}

    \vspace{0.4cm}

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Análise Tradicional}
                \footnotesize
                \begin{itemize}
                    \item Tabela de medalhas por país
                    \item Ranking de atletas por contagem
                    \item \textit{Perde:} hierarquias, eras, relações
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Análise de Redes}
                \footnotesize
                \begin{itemize}
                    \item Quem competiu com quem?
                    \item Quem dominou cada era?
                    \item \textit{Revela:} estruturas invisíveis
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.4cm}

    \begin{alertblock}{Hipótese}
        \centering
        Padrões de dominância, hierarquias e comunidades são \textbf{quantificáveis} através de redes complexas
    \end{alertblock}
\end{frame}

% Slide 3: Questão de Pesquisa
\begin{frame}{Questão de Pesquisa}
    \begin{center}
        \Large
        \textbf{Como a estrutura de redes competitivas difere entre esportes olímpicos?}
    \end{center}

    \vspace{0.6cm}

    \begin{itemize}
        \item Existem padrões de dominância consistentes ao longo de 120 anos?
        \item Esportes individuais vs coletivos formam estruturas diferentes?
        \item Comunidades de atletas refletem hegemonias históricas?
        \item Diferenças estruturais entre competições masculinas e femininas?
    \end{itemize}

    \vspace{0.5cm}

    \begin{center}
        \footnotesize
        \textit{Essas perguntas são impossíveis de responder com tabelas de medalhas — requerem modelagem estrutural}
    \end{center}
\end{frame}

% Slide 4: Objetivos
\begin{frame}{Objetivos}
    \begin{block}{Objetivo Geral}
        Caracterizar interações competitivas entre atletas olímpicos através de métricas de redes complexas
    \end{block}

    \vspace{0.5cm}

    \begin{block}{Objetivos Específicos}
        \begin{itemize}
            \item Construir \textbf{143 redes por modalidade} representando 6 esportes
            \item Aplicar métricas de \textbf{centralidade} — identificar atletas mais influentes
            \item Detectar \textbf{comunidades} — revelar eras e hegemonias históricas
            \item Analisar diferenças estruturais entre modalidades e gêneros
            \item Disponibilizar resultados em \textbf{dashboard interativo} acessível
        \end{itemize}
    \end{block}
\end{frame}

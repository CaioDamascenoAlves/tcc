% Slide 2: Motivação e Contexto
\begin{frame}{Motivação e Contexto}
    \begin{block}{120 anos de história olímpica (1896--2016)}
        \begin{itemize}
            \item \textbf{135.571 atletas} competindo em \textbf{66 modalidades}
            \item Análises tradicionais focam em \textit{contagens de medalhas}
            \item \textbf{Por trás dos resultados:} redes complexas de relações competitivas
        \end{itemize}
    \end{block}

    \vspace{0.5cm}

    \begin{block}{Hipótese}
        Padrões de dominância, hierarquias e comunidades podem ser revelados através da \textbf{teoria de redes complexas}
    \end{block}
\end{frame}

% Slide 3: Questão de Pesquisa
\begin{frame}{Questão de Pesquisa}
    \begin{center}
        \Large
        \textbf{Como a estrutura de redes competitivas difere entre esportes olímpicos?}
    \end{center}

    \vspace{0.8cm}

    \begin{itemize}
        \item Existem padrões de dominância consistentes ao longo de 120 anos?
        \item Esportes individuais vs coletivos formam estruturas diferentes?
        \item Comunidades de atletas refletem hegemonias históricas?
        \item Diferenças entre competições masculinas e femininas?
    \end{itemize}
\end{frame}

% Slide 4: Objetivos
\begin{frame}{Objetivos}
    \begin{block}{Objetivo Geral}
        Caracterizar interações competitivas entre atletas olímpicos através de métricas de redes complexas
    \end{block}

    \vspace{0.5cm}

    \begin{block}{Objetivos Específicos}
        \begin{itemize}
            \item Construir \textbf{123 redes per-event} representando 6 esportes
            \item Aplicar métricas de \textbf{centralidade} (PageRank, Betweenness, Degree)
            \item Detectar \textbf{comunidades} e analisar modularidade
            \item Analisar diferenças estruturais entre modalidades e gêneros
            \item Validar descobertas com estatísticas históricas conhecidas
        \end{itemize}
    \end{block}
\end{frame}

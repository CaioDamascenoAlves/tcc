% Slide 2: Motivação + Questão (fundidos)
\begin{frame}{Motivação e Questão de Pesquisa}
    \begin{block}{O que as tabelas de medalhas não contam}
        \footnotesize
        120 anos · 135.571 atletas · 66 modalidades --- contagens por país perdem \textbf{hierarquias}, \textbf{eras} e \textbf{relações competitivas}
    \end{block}

    \vspace{0.2cm}

    \begin{alertblock}{\centering Questão Central}
        \centering
        \textbf{Como a estrutura de redes competitivas difere entre esportes olímpicos?}
    \end{alertblock}

    \vspace{0.2cm}

    \begin{block}{Perguntas Específicas}
        \footnotesize
        \begin{itemize}
            \item Esportes individuais vs coletivos formam estruturas diferentes?
            \item Padrões de dominância são consistentes ao longo de 120 anos?
            \item Comunidades de atletas refletem hegemonias históricas?
            \item Há diferenças estruturais entre competições masculinas e femininas?
        \end{itemize}
    \end{block}
\end{frame}

% Slide 3: O que foi construído para responder
\begin{frame}{O que foi construído para responder}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Modelagem}
                \footnotesize
                \begin{itemize}
                    \item \textbf{143 redes} independentes por modalidade
                    \item \textbf{6 esportes}, 3 tipologias competitivas
                    \item \textbf{12 casos de estudo} via \textit{Iconic Score}
                \end{itemize}
            \end{block}

            \vspace{0.2cm}

            \begin{block}{Métricas Aplicadas}
                \footnotesize
                \begin{itemize}
                    \item PageRank · Betweenness · Degree
                    \item Louvain (detecção de comunidades)
                    \item Gini (desigualdade geográfica)
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Entregáveis}
                \footnotesize
                \begin{itemize}
                    \item Dashboard interativo\\(9.510 atletas navegáveis)
                    \item Validação estatística\\(Wilcoxon · bootstrap $R^2{=}0.91$)
                    \item Análise cross-esporte inédita\\em 120 anos de Olimpíadas
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

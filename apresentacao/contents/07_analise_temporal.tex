\begin{frame}{Análise Temporal}
    \framesubtitle{Evolução de Diversidade Competitiva}
    \begin{block}{Objetivo}
        Capturar dinâmicas temporais de diversidade competitiva através das eras olímpicas.
    \end{block}

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Métrica Utilizada:}
            \begin{itemize}
                \item \textbf{Entropia de Shannon}
                \begin{itemize}
                    \footnotesize
                    \item Mede diversidade temporal
                    \item $H = -\sum_{i=1}^{n} p_i \log_2(p_i)$
                    \item Alta entropia = alta diversidade
                    \item Indica participação distribuída em múltiplos períodos
                \end{itemize}
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Períodos Analisados:}
            \begin{itemize}
                \item Era Pioneira (1896-1936)
                \item Era Pós-Guerra (1948-1980)
                \item Era Moderna (1984-2000)
                \item Era Contemporânea (2004-2021)
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Diversidade Temporal (Entropia de Shannon)}
    \framesubtitle{Persistência Competitiva ao Longo das Eras}
    \begin{block}{Interpretação}
        Alta entropia indica que atletas da comunidade participaram de forma distribuída em múltiplos períodos olímpicos (longevidade competitiva).
    \end{block}

    \begin{table}[h]
        \centering
        \footnotesize
        \begin{tabular}{lccc}
            \toprule
            \textbf{Esporte} & \textbf{Entropia Média} & \textbf{Min} & \textbf{Max} \\
            \midrule
            Swimming & 1.60 & 0.58 & 2.81 \\
            Basketball & 2.46 & 2.00 & 2.76 \\
            Football & 2.53 & 2.15 & 2.85 \\
            \bottomrule
        \end{tabular}
    \end{table}

    \vspace{0.1cm}
    \begin{block}{Achados}
        \footnotesize
        \begin{itemize}
            \item Esportes coletivos: maior diversidade temporal (2.46-2.53)
            \item Natação: diversidade moderada (1.60), maior variação entre comunidades
            \item Alta entropia reflete persistência competitiva multi-era
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Padrões de Diversidade Temporal}
    \framesubtitle{Diferenças entre Esportes Individuais e Coletivos}
    \begin{alertblock}{Insight Principal}
        \footnotesize
        Esportes coletivos apresentam comunidades que agregam atletas de múltiplas décadas devido à natureza densamente conectada destas redes, resultando em maior entropia temporal.
    \end{alertblock}

    \vspace{0.2cm}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Natação (Individual):}
            \begin{itemize}
                \footnotesize
                \item Entropia moderada (1.60)
                \item Comunidades especializadas
                \item Maior segregação temporal
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Basketball/Football:}
            \begin{itemize}
                \footnotesize
                \item Entropia elevada (2.46-2.53)
                \item Comunidades multi-era
                \item Alta integração temporal
            \end{itemize}
        \end{column}
    \end{columns}

    \vspace{0.2cm}
    \begin{block}{Conclusão}
        \footnotesize
        A estrutura do esporte (individual vs coletivo) determina padrões de diversidade temporal nas comunidades detectadas.
    \end{block}
\end{frame}

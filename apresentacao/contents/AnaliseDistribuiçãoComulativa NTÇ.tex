\begin{frame}{Função de Distribuição Cumulativa (CDF) - Natação Feminina}
    \begin{columns}
        \column{0.5\textwidth}
        \centering
        \includegraphics[width=0.95\textwidth]{figures/ADC/swimming_f_edge_cdf.png}
        \vspace{0.2cm}
        \textbf{CDF - Natação Feminina}
        \column{0.5\textwidth}
        \footnotesize
        \begin{block}{}
            \begin{itemize}
            \item Distribuição fortemente concentrada em valores baixos (0-15), com crescimento muito rápido entre 0-15. Apresenta cauda longa à direita, com valores excepcionais até 80.
        \end{itemize}
        \end{block}
    \end{columns}
    \vspace{0.2cm}
    \hrulefill
    \vspace{0.1cm}
    \footnotesize \textit{CDF: Representa a probabilidade acumulada para cada valor de peso - Natação Feminina.}
\end{frame}

\begin{frame}{Função de Distribuição Cumulativa (CDF) - Natação Masculina}
    \begin{columns}
        \column{0.5\textwidth}
        \centering
        \includegraphics[width=0.95\textwidth]{figures/ADC/swimming_m_edge_cdf.png}
        \vspace{0.2cm}
        \textbf{CDF - Natação Masculina}
        \column{0.5\textwidth}
        \footnotesize
        \begin{block}{}
        \begin{itemize}
            \item Distribuição com padrão similar ao Swimming Feminino, mas com crescimento ligeiramente mais gradual. Concentração em valores baixos (0-10) e presença de valores extremos, porém menos pronunciados que no feminino.
        \end{itemize}
        \end{block}
    \end{columns}
    \vspace{0.2cm}
    \hrulefill
    \vspace{0.1cm}
    \footnotesize \textit{CDF: Exibe a probabilidade acumulada para cada valor de peso - Natação Masculina.}
\end{frame}

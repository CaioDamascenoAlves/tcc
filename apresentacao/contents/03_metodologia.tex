% Slide 6: Modelagem Per-Event
\begin{frame}{Modelagem Per-Event}
    \begin{block}{Chave Composta: (Year, Event\_Normalized)}
        \begin{itemize}
            \item \textbf{123 redes independentes} preservando homogeneidade competitiva
            \item Exemplo: \texttt{Basketball Men's Basketball (1936, 1948, ..., 2016)}
            \item Evita misturar eventos incomparáveis (100m vs Maratona)
        \end{itemize}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{Sistema de Ponderação de Arestas}
        \begin{itemize}
            \item \textbf{Ouro:} 5 pontos
            \item \textbf{Prata:} 3 pontos
            \item \textbf{Bronze:} 2 pontos
            \item Arestas bidirecionais somando pesos entre pares de atletas
        \end{itemize}
    \end{block}
\end{frame}

% Slide 7: 6 Modalidades Analisadas
\begin{frame}{6 Modalidades Analisadas}
    \begin{block}{Tipologias Esportivas}
        \begin{description}
            \item[\textbf{Esportes Individuais (Performance):}] Athletics, Swimming
            \begin{itemize}
                \item Corridas, saltos, lançamentos, natação
                \item Competição direta por tempo/distância
            \end{itemize}

            \item[\textbf{Esportes Coletivos:}] Basketball, Football
            \begin{itemize}
                \item Times inteiros no pódio
                \item Alta densidade de conexões esperada
            \end{itemize}

            \item[\textbf{Esportes Individuais (Combate):}] Judo, Boxing
            \begin{itemize}
                \item Confronto direto 1v1
                \item Categorias de peso segregando competições
            \end{itemize}
        \end{description}
    \end{block}

    \vspace{0.2cm}
    \begin{center}
        \textbf{Iconic Score} selecionou 12 redes para análise detalhada (2 por esporte, M/F)
    \end{center}
\end{frame}

% Slide 8: Métricas de Rede
\begin{frame}{Métricas de Rede}
    \begin{block}{Centralidades}
        \begin{description}
            \item[\textbf{PageRank:}] Importância recursiva -- atletas conectados a outros importantes
            \item[\textbf{Betweenness:}] Atletas-ponte entre comunidades
            \item[\textbf{Degree Centrality:}] Número direto de conexões competitivas
        \end{description}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{Detecção de Comunidades}
        \begin{itemize}
            \item \textbf{Algoritmo Louvain} -- maximiza modularidade
            \item Revela agrupamentos naturais (eras históricas, hegemonias regionais)
            \item \textbf{Modularidade (Q):} mede separação entre comunidades
        \end{itemize}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{Estrutura de Rede}
        \begin{itemize}
            \item \textbf{Densidade:} proporção de conexões realizadas vs possíveis
            \item \textbf{Gini Coefficient:} desigualdade de distribuição de medalhas
        \end{itemize}
    \end{block}
\end{frame}

% Slide 9: Validações Metodológicas
\begin{frame}{Validações Metodológicas}
    \begin{block}{1. Louvain vs Infomap}
        \begin{itemize}
            \item Comparação em 12 redes de estudo
            \item \textbf{NMI médio = 0.97} (concordância quase perfeita)
            \item Decisão: Louvain (eficiência + interpretabilidade)
        \end{itemize}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{2. Análise de Sensibilidade de Pesos}
        \begin{itemize}
            \item Testados 6 sistemas de ponderação (3-2-1, 4-3-2, 5-3-2, etc.)
            \item \textbf{Kendall $\tau \approx 1.0$} -- rankings de PageRank estáveis
            \item Sistema 5-3-2 é robusto
        \end{itemize}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{3. Testes Estatísticos}
        \begin{itemize}
            \item Wilcoxon signed-rank test para densidade F vs M
            \item \textbf{p = 0.031} (efeito significativo)
        \end{itemize}
    \end{block}
\end{frame}

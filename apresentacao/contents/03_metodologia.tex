            \item \textbf{Representatividade estrutural:} Cobertura de tipos distintos de interdependência competitiva

        \end{itemize}
    \end{block}

    \begin{block}{Modalidades Selecionadas}

        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Modelagem de Redes Competitivas}
    \framesubtitle{Construção de Grafos Direcionados e Ponderados}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Componentes do Grafo}
                \small
                \textbf{Nós (Vértices):}
                \begin{itemize}
                    \footnotesize
                    \item Atletas medalhistas
                    \item Atributos: nome, país, esporte, medalhas
                \end{itemize}

                \vspace{0.2cm}
                \textbf{Arestas (Conexões):}
                \begin{itemize}
                    \footnotesize
                    \item Direcionadas: perdedor $\rightarrow$ vencedor
                    \item \textcolor{red}{\textbf{Conectam atletas do mesmo pódio na mesma olimpíada}}
                    \item Agrupamento: (Ano, Evento, Sexo)
                    \item Previnem arestas fantasmas entre atletas que nunca competiram juntos
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Sistema de Ponderação}
                \small
                \textbf{Pesos das Arestas:}
                \begin{itemize}
                    \footnotesize
                    \item Ouro $\leftarrow$ Prata: peso \textbf{3}
                    \item Ouro $\leftarrow$ Bronze: peso \textbf{5}
                    \item Prata $\leftarrow$ Bronze: peso \textbf{2}
                \end{itemize}

                \vspace{0.2cm}
                \textbf{Justificativa:}
                \begin{itemize}
                    \footnotesize
                    \item Maior diferença $=$ maior peso
                    \item Reflete hierarquia de performance
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.2cm}
    \begin{alertblock}{Modelagem de Rivalidades Diretas}
        \footnotesize
        \centering
        Cada pódio é único: apenas atletas que \textbf{realmente competiram juntos} são conectados
    \end{alertblock}
\end{frame}

\begin{frame}{Tratamento de Casos Especiais}
    \framesubtitle{Atletismo, Judo e Boxe}
    \begin{block}{Desafio: Modalidades Heterogêneas}
        \begin{itemize}
            \item \textbf{Atletismo:} Velocista 100m $\neq$ Saltador em vara
            \item \textbf{Judo/Boxe:} Categoria 60kg $\neq$ Categoria 100kg
        \end{itemize}
    \end{block}

    \vspace{0.3cm}
    \begin{block}{Solução Natural}
        \small
        O agrupamento \textbf{(Ano, Evento, Sexo)} já resolve automaticamente:
        \begin{itemize}
            \footnotesize
            \item Cada evento olímpico é específico: \textit{"Athletics Men's 100m"}, \textit{"Judo Men's -73kg"}
            \item Atletas de modalidades/categorias diferentes \textbf{nunca} competem no mesmo pódio
            \item Redes naturalmente separadas por especialidade
        \end{itemize}
    \end{block}

    \vspace{0.2cm}
    \begin{alertblock}{Vantagem}
        \footnotesize
        \centering
        Modelagem preserva \textbf{rivalidades reais} sem necessidade de tratamento especial
    \end{alertblock}
\end{frame}

\begin{frame}{Segmentação de Redes}
    \framesubtitle{Refinamento Analítico}
    \begin{block}{Esportes Mistos (Swimming, Athletics)}
        \begin{itemize}
            \small
            \item \textbf{Rede Individual:} Provas individuais (100m livre, salto em distância)
            \item \textbf{Rede Coletiva:} Revezamentos (4x100m, 4x400m)
            \item Análise separada permite identificar padrões distintos
        \end{itemize}
    \end{block}

    \vspace{0.3cm}
    \begin{block}{Esportes Exclusivos}
        \begin{itemize}
            \small
            \item \textbf{Individual Only:} Judo, Boxing -- apenas competições individuais
            \item \textbf{Team Only:} Basketball, Football -- apenas competições coletivas
        \end{itemize}
    \end{block}
\end{frame}

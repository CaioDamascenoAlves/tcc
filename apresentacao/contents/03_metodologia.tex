\begin{frame}{Seleção dos Esportes para Análise}
    \framesubtitle{Critérios Metodológicos}
    \begin{block}{Três Critérios Fundamentais}
        \begin{itemize}
            \item \textbf{Densidade competitiva:} Número suficiente de atletas medalhistas para análise robusta
            \item \textbf{Representatividade estrutural:} Cobertura de tipos distintos de interdependência competitiva
            \item \textbf{Objetividade de resultados:} Resultados objetivos (pódio) sem avaliação subjetiva
        \end{itemize}
    \end{block}

    \vspace{0.3cm}
    \begin{block}{Modalidades Selecionadas}
        \begin{itemize}
            \item \textbf{Swimming:} Esporte misto (individual + revezamentos) -- 1.781 atletas
            \item \textbf{Basketball:} Esporte coletivo genuíno -- 915 atletas
            \item \textbf{Football:} Esporte coletivo genuíno -- 1.568 atletas
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Modelagem de Redes Competitivas}
    \framesubtitle{Construção de Grafos Direcionados e Ponderados}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Componentes do Grafo}
                \small
                \textbf{Nós (Vértices):}
                \begin{itemize}
                    \footnotesize
                    \item Atletas medalhistas
                    \item Atributos: nome, país, esporte, medalhas
                \end{itemize}

                \vspace{0.2cm}
                \textbf{Arestas (Conexões):}
                \begin{itemize}
                    \footnotesize
                    \item Direcionadas: perdedor $\rightarrow$ vencedor
                    \item Conectam atletas do mesmo pódio
                    \item Acumulam quando compartilham múltiplos pódios
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{Sistema de Ponderação}
                \small
                \textbf{Pesos das Arestas:}
                \begin{itemize}
                    \footnotesize
                    \item Ouro $\leftarrow$ Prata: peso \textbf{3}
                    \item Ouro $\leftarrow$ Bronze: peso \textbf{5}
                    \item Prata $\leftarrow$ Bronze: peso \textbf{2}
                \end{itemize}

                \vspace{0.2cm}
                \textbf{Justificativa:}
                \begin{itemize}
                    \footnotesize
                    \item Maior diferença $=$ maior peso
                    \item Reflete hierarquia de performance
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

    \vspace{0.2cm}
    \begin{alertblock}{Resultado}
        \footnotesize
        \centering
        Atletas de sucesso tornam-se \textbf{hubs centrais} recebendo conexões de competidores que derrotaram
    \end{alertblock}
\end{frame}

\begin{frame}{Acumulação de Pesos e Segmentação}
    \framesubtitle{Refinamento da Modelagem}
    \begin{block}{Acúmulo e Segmentação}
        \begin{itemize}
            \item \textbf{Acumulação de Pesos:} Pesos das arestas aumentam quando atletas compartilham pódios em múltiplos eventos.
            \item \textbf{Segmentação:} Grafos são divididos por esporte e sexo para análises específicas.
        \end{itemize}
    \end{block}
\end{frame}

    \vspace{0.1cm}

    \begin{block}{\textit{Iconic Score} \cite{TCCRepositorio2025} --- por que 12 casos de estudo?}
        \footnotesize
        $IS_e = 0.30 \cdot L_{\text{norm}} + 0.30 \cdot G_{\text{norm}} + 0.20 \cdot V_{\text{norm}} + 0.20 \cdot B(e)$\\[0.1cm]
        \textbf{L:} longevidade (edições olímpicas) \quad \textbf{G:} diversidade geográfica (países)\\
        \textbf{V:} volume de medalhistas \quad \textbf{B:} equilíbrio M/F\\[0.1cm]
        Ex.: na Natação, \textit{100m Livre} supera 50m em L, G e V.
    \end{block}
\end{frame}

% Slide: O que é uma Rede Competitiva?
\begin{frame}[shrink=3]{O que é uma Rede Competitiva?}
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \begin{block}{Elementos da Rede}
                \begin{description}
                    \item[\textbf{Nó}] cada atleta medalhista
                    \item[\textbf{Aresta}] dois atletas venceram \textit{a mesma edição olímpica}
                    \item[\textbf{Peso}] ouro=5, prata=3, bronze=2
                    \item[\textbf{Direção}] superior $\rightarrow$ inferior
                \end{description}
            \end{block}

            \vspace{0.3cm}

            \begin{block}{Exemplo --- Atletismo 100m M, 1936}
                \footnotesize
                \begin{itemize}
                    \item Owens \textbf{(ouro)} $\rightarrow$ Metcalfe \textbf{(prata)}
                    \item Owens \textbf{(ouro)} $\rightarrow$ Osendarp \textbf{(bronze)}
                    \item Metcalfe \textbf{(prata)} $\rightarrow$ Osendarp \textbf{(bronze)}
                    \item \textit{Rede cresce: 1896 a 2016}
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.44\textwidth}
            \begin{center}
                \includegraphics[width=\textwidth]{../monografia/img/atletismo_M_100m.png}\\
                \vspace{0.2cm}
                \footnotesize
                \textbf{Atletismo 100m M} (1896--2016)\\
                \vspace{0.1cm}
                \textit{Cor = comunidade (era histórica)}\\
                \textit{Tamanho nó = PageRank}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

% Slide: Modelagem por Modalidade
\begin{frame}{Modelagem por Modalidade}
    \begin{block}{O campo \texttt{Event\_Normalized} no dataset}
        \footnotesize
        Campo que padroniza o nome de cada prova olímpica:\\[0.15cm]
        \texttt{"100m Atletismo M"} \quad \texttt{"Basquete F"} \quad \texttt{"Judô -90kg M"} \quad \ldots\\[0.2cm]
        \textbf{Daqui em diante chamamos esse campo de \textit{modalidade}.}
    \end{block}

    \vspace{0.2cm}

    \begin{columns}[T]
        \begin{column}{0.54\textwidth}
            \begin{block}{Chave de Construção: (Ano, Modalidade)}
                \footnotesize
                \begin{itemize}
                    \item \textbf{143 redes independentes} preservando homogeneidade competitiva
                    \item Ex.: \texttt{Basquete M} (1936--2016) --- rede acumulada
                    \item Evita misturar provas incomparáveis (100m vs Maratona)
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.42\textwidth}
            \begin{block}{Sistema de Ponderação de Arestas}
                \footnotesize
                \begin{itemize}
                    \item \textbf{Ouro:} 5 pts \quad \textbf{Prata:} 3 pts \quad \textbf{Bronze:} 2 pts
                    \item Aresta \textbf{direcionada}: superior $\rightarrow$ inferior
                    \item Pesos acumulados (múltiplas edições)
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% Slide: 6 Modalidades Analisadas
\begin{frame}{6 Modalidades Analisadas}
    \begin{block}{Tipologias Esportivas}
        \begin{description}
            \item[\textbf{Esportes Individuais (Performance):}] Atletismo, Natação
            \begin{itemize}
                \item Corridas, saltos, lançamentos, natação livre
                \item Campo competitivo amplo --- \textit{baixa densidade esperada}
            \end{itemize}

            \item[\textbf{Esportes Coletivos:}] Basquete, Futebol
            \begin{itemize}
                \item Times inteiros no pódio
                \item Mesmos jogadores se reencontram --- \textit{alta densidade esperada}
            \end{itemize}

            \item[\textbf{Esportes Individuais (Combate):}] Judô, Boxe
            \begin{itemize}
                \item Confronto direto 1v1 por categorias de peso
                \item Categorias segregam competidores --- \textit{densidade variável}
            \end{itemize}
        \end{description}
    \end{block}

    \vspace{0.2cm}
    \begin{center}
        \footnotesize
        \textbf{Iconic Score} selecionou 12 redes para análise detalhada (2 por esporte: M e F mais representativos)
    \end{center}
\end{frame}

\begin{frame}{Seleção dos Esportes para Análise}
    \framesubtitle{Critérios Metodológicos}
    \begin{block}{Três Critérios Fundamentais}
        \begin{itemize}
            \item \textbf{Densidade competitiva:} Número suficiente de atletas medalhistas para análise robusta
            \item \textbf{Representatividade estrutural:} Cobertura de tipos distintos de interdependência competitiva
            \item \textbf{Objetividade de resultados:} Resultados objetivos (pódio) sem avaliação subjetiva
        \end{itemize}
    \end{block}

    \vspace{0.3cm}
    \begin{block}{Modalidades Selecionadas}
        \begin{itemize}
            \item \textbf{Swimming:} Esporte misto (individual + revezamentos) -- 1.781 atletas
            \item \textbf{Basketball:} Esporte coletivo genuíno -- 915 atletas
            \item \textbf{Football:} Esporte coletivo genuíno -- 1.568 atletas
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Acumulação de Pesos e Segmentação}
    \framesubtitle{Refinamento da Modelagem}
    \begin{block}{Acúmulo e Segmentação}
        \begin{itemize}
            \item \textbf{Acumulação de Pesos:} Pesos das arestas aumentam quando atletas compartilham pódios em múltiplos eventos.
            \item \textbf{Segmentação:} Grafos são divididos por esporte e sexo para análises específicas.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Dashboard Interativo}
    \framesubtitle{Ferramenta de Visualização e Exploração}
    \begin{block}{Objetivo}
        Interface web interativa (Streamlit) para exploração visual dos resultados de análise.
    \end{block}

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Funcionalidades:}
            \begin{itemize}
                \footnotesize
                \item Visualização de redes interativas
                \item Rankings customizáveis
                \item Filtros por esporte, gênero, período
                \item Gráficos de distribuições
                \item Análise de comunidades
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Tecnologias:}
            \begin{itemize}
                \footnotesize
                \item Python + Streamlit
                \item NetworkX para análise de redes
                \item Plotly para visualizações
                \item Cosmograph para grafos
                \item HTML/CSS/JavaScript
                \item Integração CSV/JSON
            \end{itemize}
        \end{column}
    \end{columns}

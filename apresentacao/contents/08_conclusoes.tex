% Slide 24: Principais Descobertas
\begin{frame}{Principais Descobertas}
    \begin{block}{1. Heterogeneidade Estrutural entre Modalidades}
        \footnotesize
        \begin{itemize}
            \item \textbf{Modularidade:} Natação (0.89) vs Basquete (0.003) --- diferença de 300×
            \item \textbf{Densidade:} Coletivos 7× mais densos que individuais
        \end{itemize}
    \end{block}

    \vspace{0.15cm}

    \begin{block}{2. Densidade Feminina $>$ Masculina (Validada Estatisticamente)}
        \footnotesize
        \begin{itemize}
            \item \textbf{5.66× maior densidade} em competições femininas
            \item \textbf{p = 0.031} (Wilcoxon) -- efeito genuíno, controlados confounders
        \end{itemize}
    \end{block}

    \vspace{0.15cm}

    \begin{block}{3. Validação com Estatísticas Históricas}
        \footnotesize
        \begin{itemize}
            \item Rankings de PageRank coincidem com campeões conhecidos
            \item Brasil (Futebol), USA (Basketball/Swimming), Japão (Judô)
            \item Redes revelam hierarquias invisíveis em análises tradicionais
        \end{itemize}
    \end{block}
\end{frame}

% Slide 25: Contribuições
\begin{frame}{Contribuições}
    \begin{block}{1. Modelagem por Modalidade}
        \footnotesize
        \begin{itemize}
            \item \textbf{143 redes independentes} -- chave (Ano, Modalidade)
            \item Sistema de ponderação 5-3-2 validado (Kendall $\tau \approx 1.0$)
        \end{itemize}
    \end{block}

    \vspace{0.1cm}

    \begin{block}{2. Caracterização Multidimensional de Comunidades}
        \footnotesize
        \begin{itemize}
            \item Hierarquia de perfis (Núcleo, Intermediária, Periferia)
            \item Entropia temporal/geográfica -- especialização em ouro na periferia
        \end{itemize}
    \end{block}

    \vspace{0.1cm}

    \begin{block}{3. Análise Cross-Esporte}
        \footnotesize
        \begin{itemize}
            \item \textbf{6 modalidades, 120 anos} -- 3 tipologias (Performance, Coletivo, Combate)
            \item Dashboard interativo acessível (9.510 atletas navegáveis)
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Limitações e Trabalhos Futuros}

        \footnotesize

            \item \textbf{Dados até 2016} -- Paris 2024 não incluído

    \begin{block}{Trabalhos Futuros}

            \item \textbf{Incluir não-medalhistas:} Redes mais completas de competição

            \item \textbf{Evolução temporal:} Snapshots por década (1896--2016)

    \end{block}

        \begin{itemize}

        \item \textbf{Insights Estruturais}
        \begin{itemize}
            \footnotesize
            \item Comunidades emergem de fatores complexos

        \end{itemize}
    \end{enumerate}

\begin{frame}{Limitações}
    \framesubtitle{Considerações para Interpretação}
    \begin{itemize}
        \item \textbf{Amostra Limitada de Modalidades}
        \begin{itemize}
            \footnotesize
            \item Análise focada em 3 esportes (5\% do programa olímpico)
            \item Generalização requer validação em outras modalidades
        \end{itemize}

        \item \textbf{Dados até 2021}
        \begin{itemize}

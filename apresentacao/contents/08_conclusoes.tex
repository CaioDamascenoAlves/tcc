% Slide 24: Principais Descobertas
\begin{frame}{Principais Descobertas}
    \begin{block}{1. Heterogeneidade Estrutural entre Modalidades}
        \footnotesize
        \begin{itemize}
            \item \textbf{Modularidade:} Natação (0.89) vs Basquete (0.003) --- diferença de 300×
            \item \textbf{Densidade:} Coletivos 7× mais densos que individuais
            \item Estrutura de rede reflete natureza da competição
        \end{itemize}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{2. Densidade Feminina $>$ Masculina (Validada Estatisticamente)}
        \footnotesize
        \begin{itemize}
            \item \textbf{5.66× maior densidade} em competições femininas
            \item \textbf{p = 0.031} (Wilcoxon test) -- efeito significativo
            \item Controlados confounders (tamanho, longevidade, tipologia)
            \item \textbf{Efeito genuíno}, não artefato metodológico
        \end{itemize}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{3. Validação com Estatísticas Históricas}
        \footnotesize
        \begin{itemize}
            \item Rankings de PageRank coincidem com campeões conhecidos
            \item Brasil (Futebol), USA (Basketball/Swimming), Japão (Judô)
            \item Redes revelam hierarquias invisíveis em análises tradicionais
        \end{itemize}
    \end{block}
\end{frame}

% Slide 25: Contribuições
\begin{frame}{Contribuições}
    \begin{block}{1. Modelagem Per-Event}
        \footnotesize
        \begin{itemize}
            \item \textbf{143 redes independentes} preservando homogeneidade competitiva
            \item Chave (Year, Event\_Normalized) evita comparações incomparáveis
            \item Sistema de ponderação 5-3-2 validado (Kendall $\tau \approx 1.0$)
        \end{itemize}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{2. Caracterização Multidimensional de Comunidades}
        \footnotesize
        \begin{itemize}
            \item Hierarquia de perfis (Núcleo, Intermediária, Periferia)
            \item Entropia temporal e geográfica
            \item Descoberta: especialização em ouro na periferia
        \end{itemize}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{3. Análise Cross-Esporte}
        \footnotesize
        \begin{itemize}
            \item \textbf{6 modalidades, 120 anos} de história olímpica
            \item 3 tipologias (Performance, Coletivo, Combate)
            \item Dashboard interativo acessível (9.510 atletas navegáveis)
        \end{itemize}
    \end{block}
\end{frame}

% Slide 26: Limitações e Trabalhos Futuros
\begin{frame}{Limitações e Trabalhos Futuros}
    \begin{block}{Limitações}
        \footnotesize
        \begin{itemize}
            \item \textbf{Dados até 2016} -- Paris 2024 não incluído
            \item \textbf{Apenas medalhistas} -- não-medalhistas excluídos
            \item \textbf{12 casos detalhados} --- 143 redes construídas, análise aprofundada em 12
        \end{itemize}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{Trabalhos Futuros}
        \footnotesize
        \begin{enumerate}
            \item \textbf{Expandir modalidades:} Incluir esportes artísticos (Ginástica, Patinação)
            \item \textbf{Incluir não-medalhistas:} Redes mais completas de competição
            \item \textbf{Análise multicamada:} Combinar dimensões socioeconômicas
            \begin{itemize}
                \footnotesize
                \item Camada 1: Competição direta (atual)
                \item Camada 2: PIB/investimento esportivo por país
                \item Camada 3: Idade, altura, peso (atributos físicos)
            \end{itemize}
            \item \textbf{Evolução temporal:} Snapshots por década (1896--1920, 1920--1940, etc.)
            \item \textbf{Redes bipartidas:} Atletas $\leftrightarrow$ Eventos (versatilidade)
        \end{enumerate}
    \end{block}
\end{frame}

% Slide 23: Arquitetura de Publicação
\begin{frame}{Arquitetura de Publicação}
    \begin{block}{Desafio: Limite de 1GB RAM no Streamlit Cloud}
        \footnotesize
        NetworkX com 9.378 nós excede memória gratuita disponível
    \end{block}

    \vspace{0.3cm}

    \begin{block}{Solução: Google Drive + Streamlit Cloud}
        \footnotesize
        \begin{enumerate}
            \item \textbf{Pré-processamento:} Grafos calculados localmente
            \item \textbf{Serialização:} JSON/CSV para Google Drive (público)
            \item \textbf{Fetch on-demand:} Streamlit baixa dados via HTTP
            \item \textbf{Renderização:} vis-network no navegador (cliente)
        \end{enumerate}
    \end{block}

    \vspace{0.3cm}

    \begin{block}{Trade-off}
        \footnotesize
        \begin{itemize}
            \item \textbf{Latência:} +2-3s para carregar rede (aceitável)
            \item \textbf{Acessibilidade:} Gratuito, sem servidor próprio
            \item \textbf{Escalabilidade:} Google Drive CDN global
        \end{itemize}
    \end{block}

    \vspace{0.3cm}

    \begin{alertblock}{URL}
        \footnotesize
        \centering
        \textbf{dashboard-tcc.streamlit.app}
    \end{alertblock}
\end{frame}

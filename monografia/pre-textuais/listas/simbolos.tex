\begin{simbolos}
  \item[$ G = (V, E) $] Grafo composto por conjunto de vértices $V$ e arestas $E$
  \item[$ w(e) $] Peso da aresta $e$ em um grafo ponderado
  \item[$ PR(v) $] PageRank do vértice $v$
  \item[$ d $] Fator de amortecimento do PageRank (tipicamente 0.85)
  \item[$ k $] Grau de um nó (número de conexões)
  \item[$ k_{in} $] Grau de entrada (número de arestas entrantes)
  \item[$ k_{out} $] Grau de saída (número de arestas saintes)
  \item[$ Q $] Modularidade da rede
  \item[$ C_i $] Comunidade $i$ em uma partição da rede
  \item[$ H $] Entropia de Shannon
  \item[$ \alpha $] Nível de significância estatística
  \item[$ \rho $] Coeficiente de correlação
\end{simbolos}
